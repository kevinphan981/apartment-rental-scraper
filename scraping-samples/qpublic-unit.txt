    function isNumber(evt, obj, numLen) {
        evt = (evt) ? evt : window.event;
        var charCode = (evt.which) ? evt.which : evt.keyCode;

        if ((charCode == 46 && obj.value.indexOf('.') > 0) ||
            (charCode != 46 && charCode > 31 && (charCode < 48 || charCode > 57)))
            return false;

        // Throw out anything that will create an overflow...AL
        if (obj.value >= 99999999)
            return false;

        // If specififed, throw out the number lenghts that exeed the defined amount...AL
        if (numLen && obj.value.toString().length > numLen - 1)
            return false;

        return true;
